cmake_minimum_required(VERSION 3.16)
project(SeniorSphere LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
find_library(SODIUM_LIBRARY sodium REQUIRED)
file(GLOB SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(seniorsphere ${SOURCES})
target_include_directories(seniorsphere PRIVATE ${SODIUM_INCLUDE_DIR})
target_link_libraries(seniorsphere PRIVATE ${SODIUM_LIBRARY} Threads::Threads)
if(WIN32)
  target_link_libraries(seniorsphere PRIVATE ws2_32 crypt32)
endif()
if(MSVC)
  target_compile_options(seniorsphere PRIVATE /W4)
else()
  target_compile_options(seniorsphere PRIVATE -Wall -Wextra)
endif()
